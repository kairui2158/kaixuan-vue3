const http = require('http');
const fs = require('fs');
const CDP_PORT = 9224;
const results = [];
const ts = () => new Date().toISOString();
function getTargets() {
  return new Promise((resolve, reject) => {
    http.get('http://127.0.0.1:' + CDP_PORT + '/json', (res) => {
      let data = '';
      res.on('data', c => data += c);
      res.on('end', () => resolve(JSON.parse(data)));
    }).on('error', reject);
  });
}
